def get_env(name: str) -> str:
    value = os.environ.get(name)
    if not value:
        raise RuntimeError(f"Missing required environment variable: {name}")
    return value

core_client = CoreClient(
    env_url=get_env("SCALEKIT_ENV_URL"),
    client_id=get_env("SCALEKIT_CLIENT_ID"),
    client_secret=get_env("SCALEKIT_CLIENT_SECRET"),
)
client = ToolsClient(core_client)

    identifier="zealt-user01",
    filter=scoped_filter,
    page_size=200,
)

for scoped in response.tools:
    definition = MessageToDict(scoped.tool.definition) if scoped.tool.definition else {}
    name = definition.get("name")
    desc = definition.get("description", "")
    schema = definition.get("parameters") or definition.get("inputSchema") or definition.get("schema") or {}
    props = schema.get("properties", {}) if isinstance(schema, dict) else {}
    if "name" in props or "channel" in desc.lower() or "channel" in (name or ""):
        print("TOOL:", name, "ID:", scoped.tool.id)
        print("DESC:", desc)
        print("PROPS:", list(props.keys()))
        print("---")
