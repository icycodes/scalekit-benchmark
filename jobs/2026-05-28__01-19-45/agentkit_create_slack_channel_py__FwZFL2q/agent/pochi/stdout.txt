import os, requests, json

env_url = os.environ['SCALEKIT_ENV_URL']
client_id_env = os.environ['SCALEKIT_CLIENT_ID']
client_secret_env = os.environ['SCALEKIT_CLIENT_SECRET']

    'grant_type': 'client_credentials',
    'client_id': client_id_env,
    'client_secret': client_secret_env,
})
scalekit_token = auth_resp.json().get('access_token')
headers = {'Authorization': f'Bearer {scalekit_token}', 'Content-Type': 'application/json'}

# Check the second page of providers for SLACK
for page in range(1, 5):
    resp = requests.get(f'{env_url}/api/v1/providers', headers=headers, params={'page_size': 50, 'page_token': str(page)})
    data = resp.json()
    providers = data.get('providers', [])
    for p in providers:
        if 'SLACK' in p.get('identifier', '').upper() or 'slack' in p.get('display_name', '').lower():
            print(f'Found Slack provider: {json.dumps(p, indent=2)}')
    if not data.get('next_page_token'):
        break
    print(f'Page {page} has {len(providers)} providers, next token: {data.get(\"next_page_token\")}')
