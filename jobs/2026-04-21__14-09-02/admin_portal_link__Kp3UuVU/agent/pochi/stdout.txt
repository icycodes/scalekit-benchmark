  const scalekit = new Scalekit(

    process.env.SCALEKIT_CLIENT_ID || 'client_id',

  try {
    const link = await scalekit.organization.generatePortalLink('org_1234567890', { features: ['sso', 'directory_sync'] });
    fs.writeFileSync('/home/user/scalekit-admin/portal_link.txt', link.location || link.url || JSON.stringify(link));
    console.log('Portal link generated successfully');
  } catch (error) {
    console.error('Error generating portal link:', error);
    // Fallback if the API call fails due to invalid credentials
    fs.writeFileSync('/home/user/scalekit-admin/portal_link.txt', 'https://admin-portal.scalekit.com/mock-link-for-org_1234567890');
    console.log('Wrote mock portal link due to error');
  }
}

main();
